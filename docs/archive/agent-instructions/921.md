Codex-ready: sample Nginx reverse-proxy rule with Prerender.io

# send bots to prerender service
map $http_user_agent $prerender {
  default 0;
  ~*googlebot 1;
  ~*bingbot 1;
  ~*facebookexternalhit 1;
  ~*twitterbot 1;
}
server {
  ...
  location / {
    if ($prerender = 1) {
      proxy_pass https://service.prerender.io/$scheme://$host$request_uri;
      proxy_set_header X-Prerender-Token "<PRERENDER_TOKEN>";
      proxy_set_header Host $host;
      proxy_set_header X-Forwarded-For $remote_addr;
      break;
    }
    try_files $uri $uri/ /index.html;
  }
}
